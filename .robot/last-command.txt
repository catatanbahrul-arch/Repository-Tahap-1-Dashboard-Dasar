
set -e

SDK="$HOME/lib/android-sdk-9123335"

echo "===== BATCH 1.1 ====="
echo "SDK=$SDK"

export ANDROID_HOME="$SDK"
export ANDROID_SDK_ROOT="$SDK"
export PATH="$SDK/cmdline-tools/bin:$SDK/platform-tools:$SDK/build-tools/35.0.0:$PATH"

echo
echo "===== SDKMANAGER ====="
SDKMANAGER="$SDK/cmdline-tools/bin/sdkmanager"

if [ ! -x "$SDKMANAGER" ]; then
    echo "ERROR: sdkmanager tidak ditemukan"
    exit 20
fi

"$SDKMANAGER" --sdk_root="$SDK" --version

echo
echo "===== SDK PACKAGES ====="
"$SDKMANAGER" --sdk_root="$SDK" --list_installed 2>&1 | head -80

echo
echo "===== PLATFORM 35 ====="
test -d "$SDK/platforms/android-35"
echo "OK"

echo
echo "===== BUILD TOOLS 35 ====="
test -d "$SDK/build-tools/35.0.0"
echo "OK"

echo
echo "===== ADB ====="
command -v adb
adb version | head -3

echo
echo "===== AAPT2 ====="
command -v aapt2
aapt2 version | head -2

echo
echo "===== D8 ====="
command -v d8
d8 --version 2>&1 | head -2

echo
echo "===== APKSIGNER ====="
command -v apksigner
apksigner --version

echo
echo "===== JAVA ====="
java -version 2>&1 | head -3

echo
echo "===== GRADLE ====="
gradle --version | head -12

echo
echo "===== HASIL ====="
echo "ANDROID BUILD ENVIRONMENT VERIFIED"
