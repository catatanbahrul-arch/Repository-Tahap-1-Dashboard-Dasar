
set -e

echo "===== BATCH 3.1: GRADLE WRAPPER + MASTER UI BUILD ====="

export ANDROID_HOME="$HOME/lib/android-sdk-9123335"
export ANDROID_SDK_ROOT="$ANDROID_HOME"
export PATH="$ANDROID_HOME/cmdline-tools/bin:$ANDROID_HOME/platform-tools:$ANDROID_HOME/build-tools/35.0.0:$PATH"

echo
echo "===== CHECK PROJECT ====="
test -f settings.gradle
test -f build.gradle
test -f app/build.gradle
test -f app/src/main/AndroidManifest.xml
test -f app/src/main/java/com/robotmessenger/facebook/MainActivity.java

echo "Project source: OK"

echo
echo "===== CREATE GRADLE WRAPPER ====="

gradle --no-daemon wrapper \
    --gradle-version 8.7 \
    --distribution-type bin

test -f gradlew
chmod +x gradlew

echo "gradlew: OK"

echo
echo "===== WRAPPER VERSION ====="
./gradlew --version | head -20

echo
echo "===== BUILD MASTER UI ====="

./gradlew \
    --no-daemon \
    --max-workers=1 \
    :app:assembleDebug

echo
echo "===== BUILD EXIT ====="
echo "BUILD_EXIT_CODE=0"

echo
echo "===== APK RESULT ====="

find app/build/outputs/apk \
    -type f \
    -print \
    -exec ls -lh {} \;

echo
echo "===== SOURCE CHECK ====="

echo "--- MainActivity.java ---"
wc -l app/src/main/java/com/robotmessenger/facebook/MainActivity.java

echo
echo "--- Manifest ---"
wc -l app/src/main/AndroidManifest.xml

echo
echo "--- Styles ---"
wc -l app/src/main/res/values/styles.xml

echo
echo "===== BATCH 3.1 SUCCESS ====="
echo "MASTER UI DASHBOARD BUILD OK"
