
echo "===== BATCH 1.2: BINARY DIAGNOSTIC ====="

SDK="$HOME/lib/android-sdk-9123335"

echo
echo "===== DEVICE ARCH ====="
uname -m
getprop ro.product.cpu.abi
getprop ro.product.cpu.abilist

echo
echo "===== TERMUX ARCH ====="
dpkg --print-architecture

echo
echo "===== ADB SDK FILE ====="
file "$SDK/platform-tools/adb" 2>&1 || true
ls -lh "$SDK/platform-tools/adb" 2>&1 || true

echo
echo "===== AAPT2 SDK FILE ====="
file "$SDK/build-tools/35.0.0/aapt2" 2>&1 || true
ls -lh "$SDK/build-tools/35.0.0/aapt2" 2>&1 || true

echo
echo "===== TERMUX AAPT2 ====="
command -v aapt2
file "$(command -v aapt2)" 2>&1 || true

echo
echo "===== TERMUX D8 ====="
command -v d8
file "$(command -v d8)" 2>&1 || true

echo
echo "===== TERMUX APKSIGNER ====="
command -v apksigner
file "$(command -v apksigner)" 2>&1 || true

echo
echo "===== SDK DIRECTORY ====="
echo "$SDK"

echo
echo "===== CMDLINE TOOLS ====="
"$SDK/cmdline-tools/bin/sdkmanager" --sdk_root="$SDK" --version 2>&1 || true
