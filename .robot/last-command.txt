
set +e

echo "===== BATCH 2.1: GRADLE DIAGNOSTIC + FIX ====="

export ANDROID_HOME="$HOME/lib/android-sdk-9123335"
export ANDROID_SDK_ROOT="$ANDROID_HOME"
export PATH="$ANDROID_HOME/cmdline-tools/bin:$ANDROID_HOME/platform-tools:$ANDROID_HOME/build-tools/35.0.0:$PATH"

echo
echo "===== STOP OLD GRADLE DAEMONS ====="
gradle --stop 2>&1 || true

echo
echo "===== GRADLE CONFIG ====="
cat gradle.properties

echo
echo "===== ROOT BUILD ====="
cat build.gradle

echo
echo "===== APP BUILD ====="
cat app/build.gradle

echo
echo "===== SETTINGS ====="
cat settings.gradle

echo
echo "===== MANIFEST ====="
cat app/src/main/AndroidManifest.xml

echo
echo "===== JAVA ====="
java -version 2>&1 | head -3

echo
echo "===== SDK ====="
echo "ANDROID_HOME=$ANDROID_HOME"
test -d "$ANDROID_HOME/platforms/android-35" && echo "Platform 35: OK"
test -d "$ANDROID_HOME/build-tools/35.0.0" && echo "Build Tools 35: OK"

echo
echo "===== BUILD WITHOUT DAEMON ====="
gradle --no-daemon :app:assembleDebug --stacktrace

EXIT_CODE=$?

echo
echo "===== BUILD EXIT CODE ====="
echo "$EXIT_CODE"

if [ "$EXIT_CODE" -eq 0 ]; then
    echo
    echo "===== APK ====="
    find app/build/outputs/apk -type f -print -exec ls -lh {} \;
    echo
    echo "===== BATCH 2.1 SUCCESS ====="
else
    echo
    echo "===== BATCH 2.1 FAILED ====="
fi

exit "$EXIT_CODE"
