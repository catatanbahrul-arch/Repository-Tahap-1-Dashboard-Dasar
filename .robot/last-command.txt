
set +e

echo "===== BATCH 2.2: GRADLE/JAVA ENV DIAGNOSTIC ====="

echo
echo "===== MEMORY ====="
free -h
echo
cat /proc/meminfo | head -20

echo
echo "===== PROCESS LIMIT ====="
ulimit -a

echo
echo "===== JAVA PROCESSES ====="
ps -ef | grep -E "[j]ava|[g]radle" || true

echo
echo "===== JAVA HOME ====="
echo "JAVA_HOME=$JAVA_HOME"
command -v java
readlink -f "$(command -v java)" 2>/dev/null || true

echo
echo "===== JAVA VERSION ====="
java -version 2>&1

echo
echo "===== JAVA SIMPLE TEST ====="
cat > /tmp/TestJava.java <<'EOF'
public class TestJava {
    public static void main(String[] args) {
        System.out.println("JAVA_PROCESS_OK");
        System.out.println("Available processors: " + Runtime.getRuntime().availableProcessors());
        System.out.println("Max memory: " + Runtime.getRuntime().maxMemory());
        System.out.println("Free memory: " + Runtime.getRuntime().freeMemory());
    }
}
EOF

javac /tmp/TestJava.java 2>&1
java -cp /tmp TestJava 2>&1

echo
echo "===== GRADLE VERSION ====="
gradle --version 2>&1 | head -20

echo
echo "===== GRADLE DAEMON CONFIG ====="
find "$HOME/.gradle" -maxdepth 2 -type f \
    \( -name "gradle.properties" -o -name "*.lock" \) \
    -print 2>/dev/null | head -30

echo
echo "===== PROJECT FILES ====="
find . -maxdepth 3 -type f \
    ! -path "./.git/*" \
    ! -path "./.robot/*" \
    | sort

echo
echo "===== DIAGNOSTIC EXIT ====="
echo "DIAGNOSTIC_COMPLETE"
